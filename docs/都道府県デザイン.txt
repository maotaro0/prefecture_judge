<script>
document.addEventListener("DOMContentLoaded", () => {

  const ANSWER_COUNT = 3

  // 今は画面確認用の仮正解
  const ANSWER_IDS = [2, 3, 5]

  const hints = [
    "正解の3県のうち、2県は人口100万人以上です。",
    "正解の3県のうち、1県はりんごの収穫量が全国TOP10です。",
    "正解の3県には、九州地方の県が含まれています。",
    "正解の3県のうち、1県は日本の首都です。",
    "正解の3県は、それぞれ異なる地方にあります。"
  ]


  let selectedIds = []
  let currentHintIndex = 0


  const prefectureButtons =
    document.querySelectorAll(".prefecture-button")

  const selectedCount =
    document.getElementById("selected-count")

  const submitButton =
    document.getElementById("submit-answer")

  const resetButton =
    document.getElementById("reset-selection")

  const nextHintButton =
    document.getElementById("next-hint")

  const giveupButton =
    document.getElementById("giveup-button")

  const currentHintTitle =
    document.getElementById("current-hint-title")

  const currentHintText =
    document.getElementById("current-hint-text")

  const currentHintNumber =
    document.getElementById("current-hint-number")

  const historyContainer =
    document.getElementById("history-container")

  const resultModal =
    document.getElementById("result-modal")


  // =====================================
  // 都道府県選択
  // =====================================

  prefectureButtons.forEach(button => {

    button.addEventListener("click", () => {

      const id =
        Number(button.dataset.prefectureId)


      if (selectedIds.includes(id)) {

        selectedIds =
          selectedIds.filter(
            selectedId => selectedId !== id
          )

        button.classList.remove("selected")

      } else {

        if (selectedIds.length >= ANSWER_COUNT) {
          return
        }

        selectedIds.push(id)

        button.classList.add("selected")
      }


      updateSelection()
    })

  })


  function updateSelection() {

    selectedCount.textContent =
      selectedIds.length

    submitButton.disabled =
      selectedIds.length !== ANSWER_COUNT
  }



  // =====================================
  // リセット
  // =====================================

  resetButton.addEventListener("click", () => {
    resetSelection()
  })


  function resetSelection() {

    selectedIds = []

    prefectureButtons.forEach(button => {
      button.classList.remove("selected")
    })

    updateSelection()
  }



  // =====================================
  // 回答
  // =====================================

  submitButton.addEventListener("click", () => {

    const hitCount =
      selectedIds.filter(id =>
        ANSWER_IDS.includes(id)
      ).length


    const selectedNames =
      selectedIds.map(id => {

        return document.querySelector(
          `[data-prefecture-id="${id}"]`
        ).dataset.prefectureName

      })


    updateCurrentHistory(
      selectedNames,
      hitCount
    )


    if (hitCount === ANSWER_COUNT) {

      showResult(true, hitCount)
      return
    }


    setTimeout(() => {

      resetSelection()
      showNextHint()

    }, 700)

  })



  // =====================================
  // 次のヒント
  // =====================================

  nextHintButton.addEventListener("click", () => {

    updateSkippedHistory()

    resetSelection()

    showNextHint()
  })


  function showNextHint() {

    if (
      currentHintIndex >=
      hints.length - 1
    ) {
      return
    }


    currentHintIndex++


    currentHintTitle.textContent =
      `ヒント ${currentHintIndex + 1}`

    currentHintText.textContent =
      hints[currentHintIndex]

    currentHintNumber.textContent =
      `${currentHintIndex + 1} / ${hints.length}`


    addHistoryItem()
  }



  // =====================================
  // 履歴
  // =====================================

  function addHistoryItem() {

    document
      .querySelectorAll(".history-item")
      .forEach(item => {

        item.classList.remove("active")

      })


    const item =
      document.createElement("article")


    item.className =
      "history-item active"


    item.innerHTML = `
      <button
        type="button"
        class="history-toggle"
      >
        <span>
          💡 ヒント ${currentHintIndex + 1}（今回）
        </span>

        <span>⌃</span>
      </button>

      <div class="history-content">

        <p>
          ${hints[currentHintIndex]}
        </p>

        <div class="history-answer">

          <small>
            あなたの回答
          </small>

          <p class="waiting">
            まだ回答していません
          </p>

        </div>

      </div>
    `


    historyContainer.prepend(item)

    activateAccordion()
  }



  function updateCurrentHistory(
    names,
    hitCount
  ) {

    const activeItem =
      document.querySelector(
        ".history-item.active"
      )

    if (!activeItem) return


    const answer =
      activeItem.querySelector(
        ".history-answer"
      )


    answer.innerHTML = `
      <small>
        あなたの回答
      </small>

      <div class="answer-tags">

        ${names.map(name =>
          `<span>${name}</span>`
        ).join("")}

      </div>

      <div class="correct-count">
        正解数：
        <strong>
          ${hitCount} / ${ANSWER_COUNT}
        </strong>
      </div>
    `

  }



  function updateSkippedHistory() {

    const activeItem =
      document.querySelector(
        ".history-item.active"
      )

    if (!activeItem) return


    const answer =
      activeItem.querySelector(
        ".history-answer"
      )


    answer.innerHTML = `
      <small>
        あなたの回答
      </small>

      <p class="skipped">
        回答せず次のヒントを取得しました
      </p>
    `

  }



  // =====================================
  // アコーディオン
  // =====================================

  function activateAccordion() {

    document
      .querySelectorAll(".history-toggle")
      .forEach(toggle => {

        toggle.onclick = () => {

          const item =
            toggle.closest(
              ".history-item"
            )

          item.classList.toggle(
            "collapsed"
          )

        }

      })

  }

  activateAccordion()



  // =====================================
  // ギブアップ
  // =====================================

  giveupButton.addEventListener(
    "click",
    () => {

      showResult(false, 0)

    }
  )



  // =====================================
  // リザルト
  // =====================================

  function showResult(
    isClear,
    hitCount
  ) {

    const title =
      document.getElementById(
        "result-title"
      )

    const message =
      document.getElementById(
        "result-message"
      )

    const correctCount =
      document.getElementById(
        "result-correct-count"
      )

    const hintCount =
      document.getElementById(
        "result-hint-count"
      )


    if (isClear) {

      title.textContent =
        "ゲームクリア！"

      message.textContent =
        `${currentHintIndex + 1}個目のヒントで正解しました！`

    } else {

      title.textContent =
        "ゲーム終了"

      message.textContent =
        "今回はギブアップしました。"
    }


    correctCount.textContent =
      `${hitCount} / ${ANSWER_COUNT}`

    hintCount.textContent =
      currentHintIndex + 1


    resultModal.classList.add("show")

  }


})
</script>
